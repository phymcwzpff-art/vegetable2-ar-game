const video = document.getElementById("video"); const scoreEl = document.getElementById("score"); let score = 0; let veggies = []; let bomb = null; // 카메라 실행 navigator.mediaDevices.getUserMedia({ video: true }).then(stream => { video.srcObject = stream; }); // 랜덤 채소 const veggieList = ["🥕", "🍅", "🥒", "🥬", "🌽"]; // 채소 생성 function spawnVeggie() { const el = document.createElement("div"); el.className = "emoji"; el.innerText = veggieList[Math.floor(Math.random() * veggieList.length)]; const x = window.innerWidth / 2 + (Math.random() * 200 - 100); const y = window.innerHeight / 2 + (Math.random() * 200 - 100); el.style.left = x + "px"; el.style.top = y + "px"; el.style.transform = "scale(1)"; document.body.appendChild(el); const obj = { el, x, y, size: 1 }; veggies.push(obj); } // 계속 생성 setInterval(spawnVeggie, 1500); // 간단 손 추적 (중앙 기준) let handX = window.innerWidth / 2; let handY = window.innerHeight / 2; // 마우스로 손 대신 테스트 window.addEventListener("mousemove", (e) => { handX = e.clientX; handY = e.clientY; }); // 채소 키우기 로직 function updateVeggies() { veggies.forEach(v => { const dx = handX - v.x; const dy = handY - v.y; const dist = Math.sqrt(dx*dx + dy*dy); if (dist < 120) { v.size += 0.02; score += 1; scoreEl.innerText = score; v.el.style.transform = `scale(${v.size})`; } }); } setInterval(updateVeggies, 50);// 폭탄 생성 function spawnBomb() { if (bomb) return; const el = document.createElement("div"); el.className = "emoji"; el.innerText = "💣"; const x = Math.random() * window.innerWidth; const y = Math.random() * window.innerHeight; el.style.left = x + "px"; el.style.top = y + "px"; document.body.appendChild(el); bomb = { el, x, y }; } // 6초마다 폭탄 등장 setInterval(spawnBomb, 6000); // 폭탄 충돌 체크 (손 위치) function checkBomb() { if (!bomb) return; const dx = handX - bomb.x; const dy = handY - bomb.y; const dist = Math.sqrt(dx*dx + dy*dy); if (dist < 80) { explode(); } } setInterval(checkBomb, 30); // 펑 효과 function explode() { document.getElementById("bombMsg").style.display = "block"; setTimeout(() => { document.getElementById("bombMsg").style.display = "none"; }, 600); bomb.el.remove(); bomb = null; // 점수 감소 score = Math.max(0, score - 20); scoreEl.innerText = score; } // 폭탄 피하기 이동 (자동 흔들림) function moveBomb() { if (!bomb) return; bomb.x += (Math.random() - 0.5) * 20; bomb.y += (Math.random() - 0.5) * 20; bomb.el.style.left = bomb.x + "px"; bomb.el.style.top = bomb.y + "px"; } setInterval(moveBomb, 100);
